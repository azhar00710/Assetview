# ═══ PLATFORM ENDPOINTS ═══

GET /api/v1/platforms
  # Returns all platforms the user has access to
  # Used by: Platform selector dropdown
  Response: { platforms: [{ id, name, code, status }] }

GET /api/v1/platforms/:platformId
  # Platform detail with aggregate counts
  Response: { platform: { id, name, code, status, systemCount, pnidCount, equipmentCount, instrumentCount } }


# ═══ SYSTEM ENDPOINTS (Miller Column 1) ═══

GET /api/v1/platforms/:platformId/systems
  # Returns all systems for a platform with summary counts
  # Used by: Systems column in Miller layout
  # Uses view: v_system_summary
  Query params:
    ?search=<text>        # Filter by name or code
    ?sys_type=<type>      # Filter by system type (process|utility|safety|instrument)
  Response: {
    systems: [{
      id, name, code, sysType,
      primaryPnidCount, lineCount, equipmentCount, instrumentCount
    }]
  }


# ═══ P&ID ENDPOINTS (Miller Column 2) ═══

GET /api/v1/pnids
  # Returns P&IDs, optionally filtered by system
  # Used by: P&IDs column in Miller layout
  # KEY: Uses pnid_system junction to resolve M:N
  Query params:
    ?platform_id=<uuid>   # Required: scope to platform
    ?system_id=<uuid>     # Filter by system (shows primary + cross-refs)
    ?include_xref=true    # Include cross-referenced P&IDs (default: true)
    ?status=<status>      # Filter by document status
    ?search=<text>        # Search drawing number or title
  Response: {
    pnids: [{
      id, drawingNumber, title, revision, status, hasImage,
      primarySystemId, primarySystemCode, primarySystemType,
      lineCount, totalSystemCount, xrefSystemCount,
      isPrimary  # relative to the filtered system (if system_id provided)
    }]
  }

GET /api/v1/pnids/:pnidId
  # P&ID detail with all associated systems
  Response: {
    pnid: { id, drawingNumber, title, revision, status, hasImage, imagePath },
    systems: [{ id, code, name, sysType, isPrimary }],
    annotations: {
      equipment: [{ equipmentId, tag, xPct, yPct, wPct, hPct }],
      instruments: [{ instrumentId, tag, xPct, yPct, wPct, hPct }]
    }
  }

GET /api/v1/pnids/:pnidId/systems
  # All systems referenced by this P&ID
  Response: { systems: [{ id, code, name, sysType, isPrimary }] }


# ═══ LINE ENDPOINTS (Miller Column 3) ═══

GET /api/v1/lines
  # Returns lines, with flexible filtering
  # Used by: Lines column in Miller layout
  # KEY: Lines belong to system (ownership) but appear on P&IDs (display)
  Query params:
    ?platform_id=<uuid>   # Required: scope to platform
    ?system_id=<uuid>     # Lines OWNED by this system
    ?pnid_id=<uuid>       # Lines APPEARING on this P&ID (uses pnid_line junction)
    ?include_xref=true    # When system_id set: also show cross-ref lines from P&IDs
    ?search=<text>        # Search line number or service
  Response: {
    lines: [{
      id, lineNumber, service, nominalSize, pipeClass, material,
      designPressure, designTemperature, operatingPressure, operatingTemperature,
      ownerSystemId, ownerSystemCode, ownerSystemType,
      pnidCount, equipmentCount, instrumentCount,
      pnidNumbers,           # Array of drawing numbers this line appears on
      isContinuation,        # Only when filtered by pnid_id
      isCrossReference       # True when line's system differs from filter context
    }]
  }

GET /api/v1/lines/:lineId
  # Line detail with all P&ID appearances
  Response: {
    line: { id, lineNumber, service, nominalSize, pipeClass, material, ... },
    ownerSystem: { id, code, name, sysType },
    pnids: [{ id, drawingNumber, title, isContinuation }],
    equipment: [{ id, tag, equipmentType, description, criticality }],
    instruments: [{ id, tag, instrumentType, description, scadaTag }]
  }


# ═══ EQUIPMENT ENDPOINTS (Miller Column 4) ═══

GET /api/v1/equipment
  # Returns equipment with flexible filtering
  # Used by: Equipment column in Miller layout
  Query params:
    ?platform_id=<uuid>   # Required
    ?system_id=<uuid>     # Equipment belonging to this system
    ?pnid_id=<uuid>       # Equipment appearing on this P&ID (via lines and standalone)
    ?line_id=<uuid>       # Equipment on this specific line + standalone from line's system
    ?criticality=<level>  # high|medium|low
    ?equipment_type=<type># Christmas Tree, Choke Valve, etc.
    ?search=<text>        # Search tag, description, type
  Response: {
    equipment: [{
      id, tag, equipmentType, description,
      systemId, systemCode,
      criticality, silLevel, inspectionGroup, corrosionLoop,
      lineId, lineNumber,      # null if standalone
      isStandalone,            # true if lineId is null
      vtSceneId, vtHotspotId   # Virtual Tool references
    }]
  }

GET /api/v1/equipment/:equipmentId
  # Full equipment detail
  Response: {
    equipment: { ... all fields ... },
    system: { id, code, name },
    line: { id, lineNumber, service } | null,
    pnids: [{ id, drawingNumber, annotationPosition }],
    instruments: [{ id, tag, instrumentType }]  # Instruments on same line
  }


# ═══ INSTRUMENT ENDPOINTS (Miller Column 5) ═══

GET /api/v1/instruments
  # Returns instruments with flexible filtering
  Query params:
    ?platform_id=<uuid>
    ?system_id=<uuid>
    ?pnid_id=<uuid>       # Via pnid_line junction → line → instruments
    ?line_id=<uuid>
    ?instrument_type=<type>
    ?search=<text>        # Search tag, scada_tag, description
  Response: {
    instruments: [{
      id, tag, instrumentType, description,
      rangeMin, rangeMax, rangeUnit, scadaTag,
      systemId, systemCode,
      lineId, lineNumber
    }]
  }


# ═══ REGISTER ENDPOINTS (full table views) ═══

GET /api/v1/registers/pnids
  # Full P&ID register for a platform
  Query params:
    ?platform_id=<uuid>
    ?sort_by=<field>      # drawingNumber|title|revision|status|primarySystemCode
    ?sort_dir=asc|desc
    ?filter[field]=<value> # Per-column filtering
    ?page=1&limit=100
  Response: {
    data: [{ drawingNumber, title, revision, status, primarySystem, allSystems, systemCount }],
    total: 24, page: 1, limit: 100
  }

GET /api/v1/registers/lines
  # Full Line List register
  Query params: (same pattern)
  Response: {
    data: [{ lineNumber, service, size, pipeClass, material, dp, dt, systemCode, pnids, pnidCount }],
    total, page, limit
  }

GET /api/v1/registers/equipment
  # Full Equipment register
  Response: {
    data: [{ tag, type, description, system, criticality, sil, inspGroup, corrLoop, line }],
    total, page, limit
  }

GET /api/v1/registers/instruments
  # Full Instrument Index
  Response: {
    data: [{ tag, type, description, range, scadaTag, system, line }],
    total, page, limit
  }


# ═══ SEARCH ENDPOINT (global) ═══

GET /api/v1/search
  # Global search across all entities
  # Uses the search_all() database function
  Query params:
    ?platform_id=<uuid>
    ?q=<text>             # Search query
    ?limit=50
  Response: {
    results: [{
      entityType,  # 'system'|'pnid'|'line'|'equipment'|'instrument'
      entityId,
      tag,         # Primary identifier
      description,
      systemCode,
      relevance    # 0-1 score
    }]
  }

POST /api/v1/annotations/:annotationId/replies

DELETE /api/v1/annotations/:annotationId
